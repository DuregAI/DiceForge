if (UnityEditor.EditorApplication.isPlaying) throw new System.InvalidOperationException("Import profiles outside Play Mode.");
string assetPath = "Assets/_Project/07_Art/DemoRCCharacters/DialogueProfiles/speaking-atlas-v1.png";
UnityEditor.AssetDatabase.ImportAsset(assetPath, UnityEditor.ImportAssetOptions.ForceSynchronousImport);
var importer = (UnityEditor.TextureImporter)UnityEditor.AssetImporter.GetAtPath(assetPath);
importer.textureType = UnityEditor.TextureImporterType.Sprite;
importer.spriteImportMode = UnityEditor.SpriteImportMode.Multiple;
importer.npotScale = UnityEditor.TextureImporterNPOTScale.None;
importer.maxTextureSize = 2048;
importer.textureCompression = UnityEditor.TextureImporterCompression.Uncompressed;
importer.alphaIsTransparency = true;
importer.mipmapEnabled = false;
importer.isReadable = true;
importer.filterMode = UnityEngine.FilterMode.Bilinear;
importer.wrapMode = UnityEngine.TextureWrapMode.Clamp;
importer.SaveAndReimport();
var texture = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Texture2D>(assetPath);
importer.GetSourceTextureWidthAndHeight(out int originalWidth, out int originalHeight);
if (texture.width != originalWidth || texture.height != originalHeight) throw new System.InvalidOperationException("Profile atlas changed size on import.");
var factory = new UnityEditor.U2D.Sprites.SpriteDataProviderFactories();
factory.Init();
var provider = factory.GetSpriteEditorDataProviderFromObject(importer);
if (provider == null) throw new System.InvalidOperationException("Sprite data provider is missing.");
provider.InitSpriteEditorDataProvider();
var capabilityProvider = provider.GetDataProvider<UnityEditor.U2D.Sprites.ISpriteFrameEditCapability>();
if (capabilityProvider == null) throw new System.InvalidOperationException("Sprite edit capability is missing.");
var capabilities = capabilityProvider.GetEditCapability();
foreach (var required in new[] { UnityEditor.U2D.Sprites.EEditCapability.CreateAndDeleteSprite, UnityEditor.U2D.Sprites.EEditCapability.EditSpriteRect, UnityEditor.U2D.Sprites.EEditCapability.EditPivot, UnityEditor.U2D.Sprites.EEditCapability.EditSpriteName })
    if (!capabilities.HasCapability(required)) throw new System.InvalidOperationException("Importer cannot edit profiles: " + required);
var previous = provider.GetSpriteRects();
string[] ids = { "tish", "luma", "bum", "koren", "shal", "ryzh" };
var rects = new UnityEditor.SpriteRect[ids.Length];
var pairs = new System.Collections.Generic.List<UnityEditor.SpriteNameFileIdPair>();
var pixels = texture.GetPixels32();
int transparent = 0;
foreach (var pixel in pixels) if (pixel.a == 0) transparent++;
if (transparent < pixels.Length / 3) throw new System.InvalidOperationException("Expected genuinely transparent speaking profiles.");
for (int i = 0; i < ids.Length; i++)
{
    int column = i % 3, row = i / 3;
    int x0 = originalWidth * column / 3, x1 = originalWidth * (column + 1) / 3;
    int y0 = row == 0 ? originalHeight / 2 : 0, y1 = row == 0 ? originalHeight : originalHeight / 2;
    int minX = x1, maxX = x0, minY = y1, maxY = y0;
    for (int y = y0; y < y1; y++)
        for (int x = x0; x < x1; x++)
            if (pixels[y * originalWidth + x].a > 24)
            { minX = System.Math.Min(minX, x); maxX = System.Math.Max(maxX, x); minY = System.Math.Min(minY, y); maxY = System.Math.Max(maxY, y); }
    if (maxX <= minX || maxY <= minY) throw new System.InvalidOperationException("Empty profile cell: " + ids[i]);
    minX = System.Math.Max(x0, minX - 4); minY = System.Math.Max(y0, minY - 4);
    maxX = System.Math.Min(x1 - 1, maxX + 4); maxY = System.Math.Min(y1 - 1, maxY + 4);
    var old = System.Array.Find(previous, r => r.name == ids[i]);
    var id = old == null ? UnityEngine.GUID.Generate() : old.spriteID;
    rects[i] = new UnityEditor.SpriteRect { name = ids[i], rect = new UnityEngine.Rect(minX, minY, maxX - minX + 1, maxY - minY + 1), alignment = UnityEngine.SpriteAlignment.Custom, pivot = new UnityEngine.Vector2(.5f, 0), spriteID = id };
    pairs.Add(new UnityEditor.SpriteNameFileIdPair(ids[i], id));
}
var nameProvider = provider.GetDataProvider<UnityEditor.U2D.Sprites.ISpriteNameFileIdDataProvider>();
if (nameProvider == null) throw new System.InvalidOperationException("Sprite name/file ID provider is missing.");
provider.SetSpriteRects(rects);
nameProvider.SetNameFileIdPairs(pairs);
provider.Apply();
importer.isReadable = false;
importer.SaveAndReimport();
var catalog = UnityEditor.AssetDatabase.LoadAssetAtPath<Diceforge.Dialogue.DemoNarrativeCatalog>("Assets/_Project/Resources/DemoRC/Narrative.asset");
int bound = 0;
foreach (var asset in UnityEditor.AssetDatabase.LoadAllAssetsAtPath(assetPath))
    if (asset is UnityEngine.Sprite sprite)
    {
        var speaker = System.Array.Find(catalog.speakers, s => s.id == sprite.name);
        if (speaker == null) throw new System.InvalidOperationException("Unknown profile speaker: " + sprite.name);
        speaker.dialogueProfile = sprite; bound++;
    }
UnityEditor.EditorUtility.SetDirty(catalog);
UnityEditor.AssetDatabase.SaveAssets();
return new { profiles = bound, originalWidth, originalHeight, transparentPixels = transparent, totalPixels = pixels.Length };
